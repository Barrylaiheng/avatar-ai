class RAGEngineLite:
    """
    RAG引擎轻量版（内联在ai_core.py中避免循环导入）
    
    完整版请使用 src/rag_engine.py
    """
    
    # 意图分类关键词（v3.1 精简版）
    KEYWORD_MAP = {
        "food_delivery": ["外卖", "饿了�?, "美团外卖", "红包"],
        "brand_deal": ["肯德�?, "麦当�?, "瑞幸", "星巴�?, "奶茶", "咖啡", "团购"],
        "movie_ticket": ["电影", "电影�?, "影院"],
        "price_compare": ["优惠", "便宜", "�?, "返利", "推荐", "多少�?],
        "travel": ["打车", "高铁", "T3"],
        "hotel": ["酒店", "住宿", "民宿"],
        "video_member": ["会员", "爱奇�?, "腾讯视频", "vip"],
        "safety_doubt": ["安全", "靠谱", "真的", "骗人", "放心", "正规"],
        "usage_guide": ["怎么�?, "在哪�?, "怎么�?, "步骤", "教程"],
        "chat": ["你好", "谢谢", "再见", "哈喽", "在吗"]
    }
    
    def __init__(self, db_path: str = None):
        self.db_path = db_path
        self._db = None
    
    @property
    def db(self):
        if self._db is None:
            import sqlite3
            if self.db_path:
                self._db = sqlite3.connect(self.db_path, check_same_thread=False)
            else:
                self._db = _get_db()
        return self._db
    
    def classify_intent(self, user_msg: str) -> Dict[str, Any]:
        """意图分类"""
        user_msg_lower = user_msg.lower()
        
        for intent_type, keywords in self.KEYWORD_MAP.items():
            matched = [kw for kw in keywords if kw in user_msg_lower]
            if matched:
                confidence = min(0.5 + len(matched) * 0.15, 0.95)
                return {
                    "intent_type": intent_type,
                    "confidence": confidence,
                    "keywords": matched
                }
        
        return {"intent_type": "others", "confidence": 0.3, "keywords": []}
    
    def retrieve_template(self, intent_type: str, scene: str = "all") -> Optional[Dict]:
        """检索话术模�?""
        try:
            cursor = self.db.execute("""
                SELECT * FROM response_templates 
                WHERE intent_type = ? AND (scene = ? OR scene = 'all')
                AND status = 'active'
                ORDER BY priority ASC
                LIMIT 1
            """, (intent_type, scene))
            row = cursor.fetchone()
            
            if row:
                # 获取列名
                columns = [desc[0] for desc in cursor.description]
                return dict(zip(columns, row))
            return None
        except Exception:
            return None
    
    def process(self, user_msg: str, context: Dict, scene: str = "mp") -> Dict[str, Any]:
        """
        RAG处理流程
        
        Returns:
            {
                "success": bool,
                "intent_type": str,
                "intent_confidence": float,
                "template_id": str,
                "template_content": dict,
                "cps_data": dict,
                "reply_text": str,
                "metadata": dict
            }
        """
        start_time = time.time()
        
        try:
            # 1. 意图分类
            intent_result = self.classify_intent(user_msg)
            intent_type = intent_result["intent_type"]
            
            # 2. 话术检�?
            template = self.retrieve_template(intent_type, scene)
            
            # 3. 生成回复（变量替换）
            reply = self._generate_reply(template, context)
            
            cost_ms = int((time.time() - start_time) * 1000)
            
            return {
                "success": True,
                "intent_type": intent_type,
                "intent_confidence": intent_result["confidence"],
                "template_id": template.get("template_id") if template else None,
                "template_content": json.loads(template.get("template_content", "{}")) if template else None,
                "cps_data": None,
                "reply_text": reply,
                "metadata": {
                    "cost_ms": cost_ms,
                    "scene": scene,
                    "keywords": intent_result.get("keywords", [])
                }
            }
        except Exception as e:
            return {
                "success": False,
                "intent_type": "error",
                "intent_confidence": 0.0,
                "template_id": None,
                "template_content": None,
                "cps_data": None,
                "reply_text": "抱歉，系统处理异常�?,
                "metadata": {"error": str(e)}
            }
    
    def _generate_reply(self, template: Optional[Dict], context: Dict) -> str:
        """生成回复（变量替换）"""
        if not template:
            return "抱歉，我暂时没有找到相关的优惠信息～"
        
        try:
            content = json.loads(template.get("template_content", "{}"))
            text = content.get("text", "")
            variables = content.get("variables", [])
            
            # 变量替换
            var_dict = {}
            for var in variables:
                if var in context:
                    var_dict[var] = str(context[var])
            
            # 默认�?
            var_dict.setdefault("platform", "淘宝闪购")
            var_dict.setdefault("keyword", "优惠商品")
            var_dict.setdefault("count", "3")
            var_dict.setdefault("link", "[链接]")
            
            for key, value in var_dict.items():
                text = text.replace(f"{{{{{key}}}}}", value)
            
            return text
        except Exception:
            return "抱歉，处理异常，请稍后再试～"

    def match_knowledge(self, user_message: str, knowledge_type: str = None) -> list:
        """按关键词匹配知识条目"""
        msg_lower = user_message.lower()
        matches = []
        for key, entry in self.knowledge_base.items():
            if knowledge_type and not key.startswith(knowledge_type):
                continue
            keywords = entry.get("keywords", [])
            if not keywords:
                matches.append((key, entry))
            else:
                for kw in keywords:
                    if kw in msg_lower:
                        matches.append((key, entry))
                        break
        return matches

    def build_rag_prompt_sections(self, user_message: str, scene: str = "wcf") -> str:
        """根据用户消息和场景构�?RAG 模式下的 Prompt 中间�?""
        sections = []

        # 1. 锚点规则（始终存在，不限制场景）
        # 注：behavior �?task_goal 已移�?prompt 角色层（base_prompt�?
        for key, entry in self.knowledge_base.items():
            if key in ("format_general", "constraint", "no_result_mode",
                       "error_case", "scene_judgment", "risk_reminder"):
                sections.append(entry["content"])

        # 2. 场景特定规则（按 channel 注入�?
        scene_key = f"{scene}_specific"
        if scene_key in self.knowledge_base:
            sections.append(self.knowledge_base[scene_key]["content"])

        # 3. 场景关键词匹配（format_dine_in / format_takeout 等）
        matched_scenes = self.match_knowledge(user_message)
        seen_content = set()
        for key, entry in matched_scenes:
            # 跳过已有内容
            if entry["content"] in seen_content:
                continue
            # 跳过渠道限定内容（已在步�?处理�?
            if "scene" in entry:
                continue
            sections.append(entry["content"])
            seen_content.add(entry["content"])

        return "\n\n".join(sections)
