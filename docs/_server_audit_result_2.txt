@app.route("/api/wcf/receive", methods=["POST"])
@_wcf_auth_required
def wcf_receive():
    """
    接收旧电�?WCF 转来的群消息

    请求�?JSON:
    {
        "msg_id": "唯一消息ID",
        "group_name": "群名�?,
        "group_wxid": "群wxid",
        "sender_wxid": "发送者wxid",
        "sender_name": "发送者昵�?,
        "content": "消息文本",
        "msg_type": "text",
        "is_at_me": false
    }

    返回:
    {"status": "ok", "msg_id": "...", "should_reply": true/false}
    """
    try:
        data = request.get_json(force=True)
        if not data:
            return jsonify({"error": "invalid json"}), 400

        msg_id = data.get("msg_id", "")
        group_name = data.get("group_name", "")
        sender_name = data.get("sender_name", "")
        sender_wxid = data.get("sender_wxid", "")
        content = (data.get("content", "") or "").strip()
        msg_type = data.get("msg_type", "text")
        is_at_me = data.get("is_at_me", False)

        print(f"[WCF] 收到消息: group={group_name}, sender={sender_name}, type={msg_type}, content={content[:50]}", flush=True)

        # 处理语音消息
        is_voice = data.get("is_voice", False)
        voice_data = data.get("voice_data", "")
        
        if msg_type == "voice" or is_voice:
            # 语音消息：ASR转文字后再处�?
            if voice_data:
                print(f"[WCF] 收到语音消息，开始ASR识别: msg_id={msg_id}", flush=True)
                threading.Thread(
                    target=_wcf_process_voice_and_reply,
                    args=(msg_id, group_name, sender_name, sender_wxid, voice_data, is_at_me, content),
                    daemon=True
                ).start()
                return jsonify({"status": "ok", "msg_id": msg_id, "should_reply": True})
            else:
                print(f"[WCF] 语音消息缺少voice_data: msg_id={msg_id}", flush=True)
                return jsonify({"status": "ok", "msg_id": msg_id, "should_reply": False})
        
        # 处理非文字消息（图片/文件/视频等）
        # OCR功能已暂停（腾讯云OCR费用不符合项目性价比），保留代码供后续复用
        if msg_type in ("image", "video", "file", "voice") or data.get("is_image") or data.get("is_video") or data.get("is_file"):
            reply = "抱歉，目前只支持文字和语音消息的交流方式，暂不支持图片、文件、视频等。您可以发送文字描述您的需求，我来帮您找优惠～"
            with _wcf_reply_lock:
                _wcf_pending_replies[msg_id] = {"reply": reply, "ts": time.time()}
            print(f"[WCF] 收到非文字消�?{msg_type})，已回复提示: msg_id={msg_id}", flush=True)
            return jsonify({"status": "ok", "msg_id": msg_id, "should_reply": True})
        
        # 只处理文本消�?
        if msg_type != "text" or not content:
            return jsonify({"status": "ok", "msg_id": msg_id, "should_reply": False})

        # 架构对齐公众号：安全壳硬规则过滤 -> LLM全权处理
        # 所有文本消息都进入AI处理流程，由LLM判断是否需要回复、怎么回复
        # is_at_me 信息传递给LLM，供AI参考判�?
        group_wxid = data.get("group_wxid", "")
        threading.Thread(
            target=_wcf_process_and_reply,
            args=(msg_id, group_name, sender_name, sender_wxid, content, is_at_me, group_wxid),
            daemon=True
        ).start()

        return jsonify({"status": "ok", "msg_id": msg_id, "should_reply": True})

    except Exception as e:
        print(f"[WCF Error] receive: {e}\n{traceback.format_exc()}", flush=True)
        return jsonify({"error": str(e)}), 500

def _is_duplicate_msg(msg_id):
    """检查消息是否已处理�?""
    if not msg_id:
        return False
    now = time.time()
    # 清理过期记录
    expired = [k for k, v in _processed_msgs.items() if now - v > _MSGID_EXPIRE]
    for k in expired:
        del _processed_msgs[k]
    if msg_id in _processed_msgs:
        return True
    _processed_msgs[msg_id] = now
    return False
